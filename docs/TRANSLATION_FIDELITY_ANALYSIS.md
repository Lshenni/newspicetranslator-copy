def _write_model(self, model: ModelIR) -> str:
    result = f".model {model.name} {model.device_type.value}"
    
    # Add parameters
    if model.parameters:
        params = " ".join(f"{k}={v}" for k, v in model.parameters.items())
        result += f" {params}"
    
    return result
